\documentclass[11pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[table]{xcolor}
\usepackage{booktabs}

\newcommand{\JEL}[1]{\par\noindent\textit{JEL classification:} #1}
\newcommand{\keywords}[1]{\par\noindent\textit{Keywords:} #1}

% ---- Topic: Filling In the Template's Metadata ----
\title{Remittances and Consumption Smoothing}
\author{Jane Economist}

\begin{document}

\maketitle

\begin{abstract}
This paper examines remittance inflows...
\end{abstract}
\JEL{F24, D14, O15}
\keywords{remittances, migration}
