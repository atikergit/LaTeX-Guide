% ---- Topic: Grouped Headers with multicolumn ----
\section*{Grouped Headers with multicolumn}
\begin{tabular}{lccc}
  \toprule
  & \multicolumn{3}{c}{Model} \\
  \cmidrule(lr){2-4}
  Variable & (1) & (2) & (3) \\
  \midrule
  Education & 0.08 & 0.07 & 0.07 \\
  \bottomrule
\end{tabular}
