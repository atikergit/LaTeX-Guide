\section{Reading Compile Errors Like a Pro}

\begin{frame}[fragile]{The Big Three Errors, Revisited}
  \begin{lstlisting}
\textb{Significant results}     % typo'd command
The share of GDP is  15%.       % unescaped percent sign
\begin{itemize} ... \end{document}
  \end{lstlisting}
  \begin{outputbox}
    \scriptsize\ttfamily
    ! Undefined control sequence. \\
    ! Missing \$ inserted. \\
    ! LaTeX Error: \textbackslash begin\{itemize\} ended by
    \textbackslash end\{document\}.
  \end{outputbox}
  \whybox{\textbf{Undefined control sequence}: almost always a typo'd
  command name -- check spelling and that the defining package is
  loaded. \textbf{Missing \$ inserted}: a math symbol (or an
  unescaped \texttt{\%}/\texttt{\&}) used outside math mode.
  \textbf{Mismatched begin/end}: count your \texttt{\textbackslash
  begin}s against your \texttt{\textbackslash end}s -- the reported
  error line is often \emph{after} the real mistake, not at it.}
\end{frame}

\section{Search Strategy: TeX StackExchange}

\begin{frame}{Finding the Fix in Under a Minute}
  \small
  \begin{enumerate}
    \item Copy the \textbf{exact} error message (drop file-specific details like line numbers).
    \item Search \texttt{tex.stackexchange.com} with that text plus the package name if relevant (e.g.\ ``natbib undefined control sequence'').
    \item Prefer answers with the \textbf{green checkmark} and a high vote count.
    \item If nothing matches, search for the \emph{symptom} instead (``booktabs table too wide'') rather than the error text.
  \end{enumerate}
  \vspace{4pt}
  Nearly every error you will ever see has already been asked and
  answered there -- searching well is a core LaTeX skill, not a
  shortcut.
\end{frame}

\section{Final Proofreading Pass}

\begin{frame}{The Pre-Submission Proofreading Checklist}
  \small
  \begin{itemize}
    \item \textbf{Spacing}: no leftover double blank lines causing odd
      paragraph breaks; consistent spacing around equations.
    \item \textbf{Orphaned headings}: a \texttt{\textbackslash section}
      stranded alone at the bottom of a page.
    \item \textbf{Table/figure placement}: nothing floated to a
      strange spot or overlapping text.
    \item \textbf{Citation style consistency}: \texttt{citet}/\texttt{citep} used correctly and consistently throughout.
    \item \textbf{All \texttt{??} resolved}: search your PDF for ``??''
      -- it means a \texttt{\textbackslash ref} or \texttt{\textbackslash cite} didn't resolve.
  \end{itemize}
\end{frame}

\section{Exporting the Final PDF}

\begin{frame}{Pre-Submission Checklist}
  \small
  \begin{itemize}
    \item Compile \textbf{twice} (or more) so all cross-references and
      the table of contents settle correctly.
    \item Confirm the PDF opens cleanly and page count matches the
      journal's length limit.
    \item Check the title, author, affiliation, abstract, and JEL
      codes match the journal's required format exactly.
    \item Download the compiled PDF from Overleaf (\texttt{Menu} $\rightarrow$ \texttt{Download PDF}) for final submission.
  \end{itemize}
\end{frame}
