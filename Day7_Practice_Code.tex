\documentclass[11pt,a4paper]{article}
\usepackage[utf8]{inputenc}

\begin{document}

% ---- Error 1: Undefined Control Sequence ----
\section*{Error 1: Undefined Control Sequence}
% BROKEN: \textb{Significant results}
% FIXED:
\textbf{Significant results}

% ---- Error 2: Missing $ Inserted ----
\section*{Error 2: Missing \$ Inserted}
% BROKEN: The share of GDP is  15%.
% FIXED (escape the percent sign):
The share of GDP is 15\%.

% ---- Error 3: Mismatched begin/end ----
\section*{Error 3: Mismatched begin/end}
% BROKEN: \begin{itemize} ... \end{document}  (itemize never closed)
% FIXED:
\begin{itemize}
  \item Every \verb|\begin| needs a matching \verb|\end|.
\end{itemize}

\end{document}
